## Implementation Examples

### Integration with React
```tsx
import { SpectrumConversationPanel } from '@unops-itg-npm/cpit-spectrum';

function CustomerSupportApp() {
  const [conversation, setConversation] = useState(supportConversation);
  
  const handleAction = (event: CustomEvent) => {
    const { action, type, value, messageId } = event.detail;
    
    switch (action) {
      case 'escalate':
        escalateToTier2(messageId);
        break;
      case 'transcript':
        emailTranscript(conversation);
        break;
    }
  };
  
  const handleSourceClick = (event: CustomEvent) => {
    const { label, value, messageId } = event.detail;
    trackResourceAccess(value, messageId);
    openExternalLink(value);
  };
  
  return (
    <SpectrumConversationPanel
      messages={JSON.stringify(conversation.messages)}
      conversationtitle={conversation.title}
      actions={JSON.stringify(conversation.actions)}
      sound={true}
      onAction={handleAction}
      onSourceClick={handleSourceClick}
      onTitleChanged={(e) => updateConversationTitle(e.detail.value)}
    />
  );
}
```

### Integration with Vue
```vue
<template>
  <spectrum-conversation-panel
    :messages="messagesJson"
    :conversationtitle="title"
    :actions="actionsJson"
    :sound="audioEnabled"
    @action="handleAction"
    @sourceClick="handleSourceClick"
    @explore="handleExploration"
  />
</template>

<script setup>
import { ref, computed } from 'vue';

const conversation = ref(educationalConversation);
const audioEnabled = ref(true);

const messagesJson = computed(() => 
  JSON.stringify(conversation.value.messages)
);

const actionsJson = computed(() => 
  JSON.stringify(conversation.value.actions)
);

const handleAction = (event) => {
  const { action, value, messageId } = event.detail;
  
  if (action === 'quiz') {
    startPracticeQuiz(messageId);
  } else if (action === 'notes') {
    saveToStudyNotes(conversation.value);
  }
};

const handleExploration = (event) => {
  const { value, messageId } = event.detail;
  addFollowUpQuestion(value, messageId);
};
</script>
```

  debugMode = environment.production ? false : true;

    return JSON.stringify(this.conversation.messages);
  }
  
  get actionsJson(): string {
    return JSON.stringify(this.conversation.actions);
  }
  
  onAction(event: CustomEvent) {
    const { action, value, messageId } = event.detail;
    
    switch (action) {
      case 'bibliography':
        this.exportBibliography();
        break;
      case 'citations':
        this.generateCitations();
        break;
    }
  }
  
  onSourceClick(event: CustomEvent) {
    const { label, value, messageId } = event.detail;
    this.bibliographyService.addSource(value, label, messageId);
    this.analyticsService.trackResourceAccess(value);
  }
